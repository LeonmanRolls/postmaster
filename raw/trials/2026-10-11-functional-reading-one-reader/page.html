Text wrap, two ways

Both versions do the same job. textWrap takes a line of text, a width and an indent, and breaks the text into lines no wider than the width, with the indent on every line after the first. A word longer than a whole line is broken. The two modules these come from pass the same tests, and a comparison on more than 280,000 inputs found no difference in their output.

Read each one as you would before changing it. Then answer the three questions at the bottom in the chat. A letter is enough, and a few words of why is welcome.

Version A

/** Python textwrap.wrap with main's settings: content width, a two-space continuation indent, * whitespace kept, long words broken, no hyphen breaks. */function textWrap(text: string, w: number, indent: string): string[] {  const chunks = text.split(WORDSEP_SIMPLE).filter((c) => c !== "");  chunks.reverse();  const lines: string[] = [];  while (chunks.length > 0) {    const curLine: string[] = [];    let curLen = 0;    const ind = lines.length > 0 ? indent : "";    const lw = w - cpLen(ind);    while (chunks.length > 0) {      const l = cpLen(chunks[chunks.length - 1]!);      if (curLen + l <= lw) {        curLine.push(chunks.pop()!);        curLen += l;      } else break;    }    if (chunks.length > 0 && cpLen(chunks[chunks.length - 1]!) > lw) {      const spaceLeft = lw < 1 ? 1 : lw - curLen;      const chunk = chunks[chunks.length - 1]!;      const [head, tail] = cpSlice(chunk, spaceLeft);      curLine.push(head);      chunks[chunks.length - 1] = tail;      curLen = curLine.reduce((a, c) => a + cpLen(c), 0);    }    if (curLine.length > 0) lines.push(ind + curLine.join(""));  }  return lines;}

Version B

/** What textWrap has so far: the lines it has closed, and the one it is filling. */interface Wrapping {  readonly lines: string[];  /** the chunks on the line being filled */  readonly open: string[];  /** their length in code points */  readonly openLen: number;} /** The line being filled, closed: with its indent unless it is the first, and not at all when empty. */function closeLine(wrapping: Wrapping, indent: string): Wrapping {  if (wrapping.open.length === 0) return wrapping;  const ind = wrapping.lines.length > 0 ? indent : "";  return { lines: [...wrapping.lines, ind + wrapping.open.join("")], open: [], openLen: 0 };} /** Put one chunk on the line being filled, closing lines as it needs. */function place(wrapping: Wrapping, chunk: string, w: number, indent: string): Wrapping {  const ind = wrapping.lines.length > 0 ? indent : "";  const room = w - cpLen(ind);  const size = cpLen(chunk);  if (wrapping.openLen + size <= room) {    return { ...wrapping, open: [...wrapping.open, chunk], openLen: wrapping.openLen + size };  }  if (size > room) {    // longer than a whole line: fill what is left of this one, and go on with the rest    const spaceLeft = room < 1 ? 1 : room - wrapping.openLen;    const [head, tail] = cpSlice(chunk, spaceLeft);    const filled = { ...wrapping, open: [...wrapping.open, head] };    return place(closeLine(filled, indent), tail, w, indent);  }  return place(closeLine(wrapping, indent), chunk, w, indent);} /** Python textwrap.wrap with main's settings: content width, a two-space continuation indent, * whitespace kept, long words broken, no hyphen breaks. */function textWrap(text: string, w: number, indent: string): string[] {  const chunks = text.split(WORDSEP_SIMPLE).filter((c) => c !== "");  const filled = chunks.reduce<Wrapping>((so, chunk) => place(so, chunk, w, indent), {    lines: [],    open: [],    openLen: 0,  });  return closeLine(filled, indent).lines;}

What both use

// textwrap's ASCII whitespace set: only these split words (U+00A0 and friends never break).const WORDSEP_SIMPLE = /([\t\n\x0b\x0c\r ]+)/u; /** Code points, as Python len() counts them. */function cpLen(s: string): number {  return [...s].length;} function cpSlice(s: string, end: number): [string, string] {  const chars = [...s];  return [chars.slice(0, end).join(""), chars.slice(end).join("")];}

The text is split into words and runs of whitespace, and both are kept. The order of A and B means nothing.
